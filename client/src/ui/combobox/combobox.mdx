## Usage

```tsx
const initialItems = ["Next.js", "Astro", "Qwik", "SolidStart", "Nuxt.js"];

return (
	<Combobox
		options={initialItems}
		placeholder="Search framework..."
		allowCustomValue>
		<ComboboxTrigger>
			<ComboboxInput>
				<ComboboxClearSelection />
			</ComboboxInput>
		</ComboboxTrigger>

		<ComboboxContent>
			{(item: string, isNewOptionValue) => (
				<ComboboxItem item={item}>
					<Show when={isNewOptionValue(item)}>+ Create {item}</Show>

					<Show when={!isNewOptionValue(item)}>{item}</Show>
				</ComboboxItem>
			)}
		</ComboboxContent>
	</Combobox>
);
```
